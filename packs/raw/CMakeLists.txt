cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0091 NEW)
project(pulse_rawpack LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
find_package(libraw CONFIG REQUIRED)
add_executable(pulse-rawpack src/rawpack.cpp)
target_include_directories(pulse-rawpack PRIVATE ../../src/common)
target_compile_definitions(pulse-rawpack PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(pulse-rawpack PRIVATE /W4 /EHsc /utf-8 /permissive-)
target_link_libraries(pulse-rawpack PRIVATE libraw::raw_r windowscodecs ole32)
target_link_options(pulse-rawpack PRIVATE /INCREMENTAL:NO /OPT:REF /OPT:ICF /DEPENDENTLOADFLAG:0xA00)
enable_testing()
add_test(NAME rawpack_formats COMMAND pulse-rawpack formats)
